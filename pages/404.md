# Page not found

This site was rebuilt in 2026 and old addresses have moved.
<span id="moved">Taking you to the [home page](/){#go} in a few seconds.</span>

Or try [research](/research/), [publications](/publications/) or
[talks and teaching](/talks/).

```{=html}
<script>
/* Send visitors from old (Hugo-era) addresses to the closest new page.
   The base path (/~aam, /newweb2 or "") comes from the <base> element the
   template sets for this page, so this works on every host. */
(function () {
  var go = document.getElementById("go");
  var base = new URL(document.baseURI).pathname.replace(/\/$/, "");
  var path = location.pathname;
  if (path.indexOf(base + "/") === 0) path = path.slice(base.length);
  var oldProjects = { alerce: "alerce", crts: "crts", edrn: "edrn", lsst: "lsst", mcl: "mcl",
                      robopol: "robopol", vifidibbs: "vifi", ztf: "ztf" };
  var m, target = "/", label = "home page";
  if ((m = path.match(/^\/project\/([^\/]+)/)) && oldProjects[m[1]]) {
    target = "/research/#" + oldProjects[m[1]]; label = "research page";
  } else if (/^\/project\//.test(path)) {
    target = "/research/"; label = "research page";
  } else if (/^\/(publication|authors|tags)\//.test(path)) {
    target = "/publications/"; label = "publications";
  } else if (/^\/(talk|teaching)\//.test(path)) {
    target = "/talks/"; label = "talks and teaching";
  } else if (/^\/personal\//.test(path)) {
    target = "/personal/"; label = "personal pages";
  }
  go.setAttribute("href", base + target);
  go.textContent = label;
  setTimeout(function () { location.replace(base + target); }, 4000);
})();
</script>
```
